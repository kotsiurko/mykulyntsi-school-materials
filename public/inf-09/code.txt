#include <Servo.h>

// Початкові налаштування
...

Servo myservo;

int lastAngle = -1;

void setup() {
  myservo.attach(pinServo);

  pinMode(pinButton1, INPUT_PULLUP);
  pinMode(pinButton2, INPUT_PULLUP);
  pinMode(pinButton3, INPUT_PULLUP);

  myservo.write(90);
}

void loop() {

  int angle = lastAngle;

  // кнопки активні при LOW (бо pullup)
  if (digitalRead(pinButton1) == LOW) angle = 0;
  else if (digitalRead(pinButton2) == LOW) angle = 90;
  else if (digitalRead(pinButton3) == LOW) angle = 160;

  // оновлюємо тільки якщо змінилось
  if (angle != lastAngle) {
    angle = constrain(angle, 5, 175);
    myservo.write(angle);
    lastAngle = angle;
  }

  delay(20);
}
